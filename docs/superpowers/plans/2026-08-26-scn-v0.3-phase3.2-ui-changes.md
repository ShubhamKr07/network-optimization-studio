# SCN v0.3, Phase 3.2 — Table Columns, Input Map, Removals, Renames — Implementation Plan

> **For agentic workers:** dispatch one agent-team task per numbered Task below (per this repo's established agent-team model, not sequential subagent-driven-development). Waves/Checkpoints are execution barriers — do not dispatch a Wave 2 task before its Checkpoint 1 gate is green on the merged state, and do not dispatch Wave 3 before Checkpoint 2 is green.

**Goal:** Ship all six changes from the approved design spec (`docs/superpowers/specs/2026-08-26-scn-v0.3-phase3.2-ui-changes-design.md`): table column additions + real postal-code data, a new pre-solve Input Map with click-to-place, removal of Reports/Compare/Reset-to-Baseline, two renames, and new tab-coverage tests.

**Architecture:** Three waves, two gate-verified checkpoints (per the spec's "Sequencing for agent-team execution" section) — this ordering exists specifically to avoid two confirmed file collisions (Track A vs. the zip-plumbing work on `openapi.yaml`; Track A vs. Track D on `Workspace.tsx`), found by checking every track's real file list against every other track's before dispatch, not assumed safe.

**Tech Stack:** Express 5 + Drizzle (api-server), React + Vite + Tailwind + Leaflet (studio), Zod (dataset-schema), Orval codegen, PuLP/CBC (unaffected — no solver changes in this phase).

```ts
// scripts/src/fetch-postal-codes.ts
//
// One-time script: reverse-geocodes every warehouse/customer/mine/station/
// refinery row's lat/lng via Nominatim and writes the result back as a new
// `zip` field on the raw dataset JSON. Resumable — skips any row that
// already has `zip` set, so a mid-run failure/rate-limit doesn't require
// starting over. Recomputes each touched package's version.json sha256
// (and bumps its version integer) at the end, since test_datasets.py
// asserts sha256 against the raw file bytes.
import { readFileSync, writeFileSync } from "fs";
import path from "path";
import { fileURLToPath } from "url";
import { createHash } from "crypto";

const __dirname = path.dirname(fileURLToPath(import.meta.url));
const REPO_ROOT = path.resolve(__dirname, "..", "..");

interface Target {
  modelId: string;
  files: string[]; // files with rows to geocode (subset of the package's full file list)
  packageFiles: string[]; // the package's FULL file list, for sha256 recomputation
}

const TARGETS: Target[] = [
  { modelId: "p-median-us", files: ["warehouses.json", "customers.json"], packageFiles: ["warehouses.json", "customers.json", "distances.json"] },
  { modelId: "transport-coal", files: ["mines.json", "stations.json"], packageFiles: ["mines.json", "stations.json", "costs.json"] },
  { modelId: "two-echelon-gold-au", files: ["refineries.json", "customers.json"], packageFiles: ["mines.json", "refineries.json", "customers.json", "distances.json"] },
];

function datasetDir(modelId: string): string {
  return path.join(REPO_ROOT, "solvers", modelId, "dataset");
}

async function reverseGeocode(lat: number, lng: number): Promise<string | null> {
  const url = `https://nominatim.openstreetmap.org/reverse?lat=${lat}&lon=${lng}&format=json`;
  const res = await fetch(url, { headers: { "User-Agent": "nos-dataset-script/1.0 (one-time postal-code sourcing)" } });
  if (!res.ok) return null;
  const body = await res.json() as { address?: { postcode?: string } };
  return body.address?.postcode ?? null;
}

function sleep(ms: number): Promise<void> {
  return new Promise((resolve) => setTimeout(resolve, ms));
}

async function processFile(modelId: string, filename: string): Promise<{ updated: number; missed: number }> {
  const filePath = path.join(datasetDir(modelId), filename);
  const raw = JSON.parse(readFileSync(filePath, "utf8")) as Record<string, { lat: number; lng: number; zip?: string }>;
  let updated = 0;
  let missed = 0;

  for (const [key, row] of Object.entries(raw)) {
    if (row.zip) continue; // already geocoded — resumable
    const zip = await reverseGeocode(row.lat, row.lng);
    if (zip) {
      raw[key] = { ...row, zip };
      updated++;
    } else {
      console.warn(`[fetch-postal-codes] no postcode for ${modelId}/${filename} row ${key} (lat=${row.lat}, lng=${row.lng})`);
      missed++;
    }
    await sleep(1000); // Nominatim usage policy: max 1 req/sec
  }

  writeFileSync(filePath, JSON.stringify(raw, null, 2) + "\n");
  return { updated, missed };
}

function recomputeVersion(modelId: string, packageFiles: string[]): void {
  const dir = datasetDir(modelId);
  const hash = createHash("sha256");
  for (const filename of [...packageFiles].sort()) {
    hash.update(readFileSync(path.join(dir, filename)));
  }
  const versionPath = path.join(dir, "version.json");
  const current = JSON.parse(readFileSync(versionPath, "utf8")) as { version: number; sha256: string };
  const next = { version: current.version + 1, sha256: hash.digest("hex") };
  writeFileSync(versionPath, JSON.stringify(next, null, 2) + "\n");
}

async function main() {
  let totalUpdated = 0;
  let totalMissed = 0;
  for (const target of TARGETS) {
    for (const filename of target.files) {
      const { updated, missed } = await processFile(target.modelId, filename);
      totalUpdated += updated;
      totalMissed += missed;
    }
    recomputeVersion(target.modelId, target.packageFiles);
    console.log(`[fetch-postal-codes] ${target.modelId}: version.json recomputed`);
  }
  console.log(`[fetch-postal-codes] done — ${totalUpdated} rows updated, ${totalMissed} rows had no postcode`);
}

main().catch((err) => {
  console.error(err);
  process.exit(1);
});
```
